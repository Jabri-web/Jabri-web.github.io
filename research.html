import numpy as np

def Z_of_t(t, xp):
    x = complex(0.5, t)
    return np.exp(-x/xp) * (x**-5) * np.log(x) * np.sin(2*np.pi / x)

def count_zeros(xp, t_max=100, step=0.02):
    ts = np.arange(0.01, t_max, step)
    ims = np.array([Z_of_t(t, xp).imag for t in ts])
    c = 0
    for i in range(1, len(ims)):
        if ims[i-1]*ims[i] < 0:
            c += 1
    return c

print("xp | 0-60 | 0-100 | ملاحظة")
for xp in [5,10,18,21,42,44,50]:
    z60 = count_zeros(xp,60)
    z100 = count_zeros(xp,100)
    note = ""
    if xp == 21: note = "⭐ تعريف"
    if xp == 42: note = "🌽 2*xp=42 يوم"
    if xp == 44: note = "🌽 مجاور"
    print(f"{xp:2} | {z60:4} | {z100:5} | {note}")
